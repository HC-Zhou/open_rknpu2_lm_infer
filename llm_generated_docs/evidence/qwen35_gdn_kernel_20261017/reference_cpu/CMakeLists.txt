cmake_minimum_required(VERSION 3.14)
project(qwen35_cpu LANGUAGES CXX)
find_package(OpenMP REQUIRED)
add_library(qwen35_cpu STATIC gdn.cpp)
target_compile_features(qwen35_cpu PRIVATE cxx_std_17)
target_compile_options(qwen35_cpu PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    # RK3588 big cores. Scheduling only: no extra ISA requirements or fast-math.
    target_compile_options(qwen35_cpu PRIVATE -mtune=cortex-a76)
endif()
target_link_libraries(qwen35_cpu PRIVATE OpenMP::OpenMP_CXX)
set_target_properties(qwen35_cpu PROPERTIES POSITION_INDEPENDENT_CODE ON)
