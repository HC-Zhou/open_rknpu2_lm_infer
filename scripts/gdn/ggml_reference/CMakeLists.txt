cmake_minimum_required(VERSION 3.14)
project(rknpu2_gdn_reference C CXX ASM)
if(NOT GGML_SOURCE_DIR)
    message(FATAL_ERROR "Set GGML_SOURCE_DIR to the pinned llama.cpp/ggml checkout")
endif()
add_subdirectory("${GGML_SOURCE_DIR}" ggml)
add_executable(gdn_reference main.cpp)
target_compile_features(gdn_reference PRIVATE cxx_std_17)
target_link_libraries(gdn_reference PRIVATE ggml ggml-cpu)
if(CPU_KERNELS_DIR)
    add_subdirectory("${CPU_KERNELS_DIR}" cpu-kernels)
    target_include_directories(gdn_reference PRIVATE "${CPU_KERNELS_DIR}")
    target_compile_definitions(gdn_reference PRIVATE GDN_HAVE_LOCAL_CPU)
    target_link_libraries(gdn_reference PRIVATE rknpu2_cpu_kernels)
endif()
